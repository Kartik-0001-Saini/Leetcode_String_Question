Question. 567) Permutation in String
Given two strings s1 and s2, return true if s2 contains a permutation of s1, or false otherwise.

In other words, return true if one of s1's permutations is the substring of s2.

solution.)
class Solution {
public:

    // Checking window is same to the string s1 or not
    bool isSameFreq(int freq[], int windowFreq[]){
        for(int i=0; i<26; i++){
            if(freq[i] != windowFreq[i]){
                return false;
            }
        }
        return true;
    }

    bool checkInclusion(string s1, string s2) {
        int n1 = s1.size(), n2 = s2.size();
        // We are taking the frequecy of each charter in s1 string
        int freq[26] = {0};
        for(int i=0; i<n1; i++){
            freq[s1[i] - 'a']++;
        }
        // Defining window size
        int windowSize = n1;
        for(int i=0; i<n2; i++){
            // making another frequency array for camparing to  the string s1
            int windowIdx = 0, idx = i;
            int windowFreq[26] = {0};
            while(windowIdx < windowSize && idx < n2){
                windowFreq[s2[idx] - 'a']++;
                windowIdx++;
                idx++;
            }

            if(isSameFreq(freq, windowFreq)){
                return true;
            }
        }

        return false;
    }
};
